<script>
/* Hides the splash. Called by the app right after its first real screen is
   drawn (and by its error screen). The app is usable at once: the splash
   stops taking touches the moment the fade starts, and is removed after it.
   Reduce Motion: no fade, it is removed at once. Holds through any slow
   connection: there is no timeout that reveals a half-loaded app. */
(function(){
  var done=false;
  function hide(){
    var el=document.getElementById("haven-splash");
    if(done||!el) return; done=true;
    try{ performance.mark("haven-splash-hide"); }catch(e){}
    var still=false; try{ still=window.matchMedia("(prefers-reduced-motion: reduce)").matches; }catch(e){}
    el.classList.add("hs-out");
    setTimeout(function(){ if(el.parentNode) el.parentNode.removeChild(el); try{ performance.mark("haven-splash-removed"); }catch(e){} }, still?0:320);
  }
  window.havenHideSplash=hide;
  // The splash is the loading screen: nothing hides it except the app.
  // A slow load gets a quiet note; files that fail to download get a clear
  // message instead of an endless splash.
  function note(text){
    var el=document.getElementById("haven-splash"); if(done||!el) return;
    var n=el.querySelector(".hs-note"); if(!n) return;
    n.textContent=text; n.classList.add("hs-show");
  }
  setTimeout(function(){ note("Still loading\u2026"); },10000);
  document.addEventListener("error",function(e){
    var t=e&&e.target; if(t&&t.tagName==="SCRIPT") note("Haven Pro couldn't finish loading. Check your connection, then reopen the app.");
  },true);
})();
</script>
